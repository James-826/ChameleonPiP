export class AdaptiveScheduler {
  private readonly queue = new RingBuffer<Task>(4096);
  public dispatch(batch: readonly Task[]): Metrics {
    const t0 = performance.now();
    for (const job of batch) this.queue.push(job);
    return { queued: batch.length, dt: performance.now() - t0 };
  }
}
